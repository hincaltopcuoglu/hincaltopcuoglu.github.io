library(plyr)

prove that joint entropy of two variables X and Y is equal to conditional entropy + entropy of Y
Prove : H(X;Y) = H(X|Y) + H(Y)

define shannon entropy

shannon_entropy <- function(x){
  df = plyr::count(x)
  n = length(x)
  df$probs = df$freq / n
  se<- -sum(df$probs * log(df$probs,base=2))
  return(se)
}
